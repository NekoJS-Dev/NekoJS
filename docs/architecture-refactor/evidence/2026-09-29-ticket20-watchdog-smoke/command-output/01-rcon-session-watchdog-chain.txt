# Ticket 20 AC4 live-watchdog smoke driver (temporary tool, embedded verbatim
# into the evidence transcript for reproducibility).
#
# Waits for the dedicated server 'Done (' marker, then drives the watchdog
# marker chain over RCON as the console sender:
#   P1  sanity: error surface + single-file reload of the good script
#   P2  Leg A: write runaway script, full reload -> candidate killed by the
#       wall-clock watchdog, old active must keep serving
#   P3  Leg A: second explicit full reload -> new candidate fails identically,
#       no auto retry, no generation bump
#   P4  Leg B: single-file reload of the runaway script -> evaluated on the
#       ACTIVE context -> watchdog isolates the active (dispatch stops)
#   P5  isolation observables: FILE reload refused, error surface
#   P6  AC4 core: full reload explicitly creates a candidate while isolated,
#       runaway still present -> candidate fails again, isolation persists,
#       observation window must show no auto recovery
#   P7  explicit recovery: remove runaway, full reload commits, markers resume
# Then stops the server cleanly.
import os
import socket
import struct
import sys
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, timeout=90.0, quiet=0.0):
        # Long overall deadline for the FIRST response packet: a reload that a
        # 3s watchdog must interrupt takes seconds before the command answers.
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        deadline = time.time() + timeout
        out = ''
        try:
            while True:
                remaining = deadline - time.time()
                if remaining <= 0:
                    break
                self.s.settimeout(remaining)
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                rid2, _ptype = struct.unpack('<ii', data[:8])
                payload = data[8:-2].decode('utf-8', 'replace')
                if rid2 == self.rid:
                    out += payload
                    if len(payload) < 4000:
                        break
        except socket.timeout:
            pass
        time.sleep(quiet)
        return out.strip()


def wait_for(path, needle, timeout):
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                if needle in f.read():
                    return True
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def stamp():
    return time.strftime('%H:%M:%S')


def main():
    log_path = sys.argv[1]
    scripts_dir = sys.argv[2]
    print('# [%s] waiting for server Done marker in %s' % (stamp(), log_path), flush=True)
    if not wait_for(log_path, 'Done (', 600):
        print('# [%s] FAIL: server never reached Done within 600s' % stamp(), flush=True)
        sys.exit(2)
    print('# [%s] server Done observed, connecting RCON 127.0.0.1:25902' % stamp(), flush=True)
    rcon = Rcon('127.0.0.1', 25902, 't20w')

    def run(label, cmd, timeout=90.0, quiet=1.5):
        resp = rcon.cmd(cmd, timeout, quiet)
        print('# [%s] RCON [%s] -> %s\n#   resp: %s' % (stamp(), label, cmd, resp if resp else '(empty)'), flush=True)
        return resp

    def observe(seconds, why):
        # Observation window: server log side (markers/kill lines) is cut later
        # from the gradle log; here we only bound the window in wall clock.
        print('# [%s] observing %ds: %s' % (stamp(), seconds, why), flush=True)
        time.sleep(seconds)
        print('# [%s] observation window ended' % stamp(), flush=True)

    good = os.path.join(scripts_dir, 't20w-good.js')
    runaway = os.path.join(scripts_dir, 't20w-runaway.js')

    print('# [%s] ==== P1: sanity on the healthy active generation ====' % stamp(), flush=True)
    run('error-healthy', 'nekojs error')
    run('file-reload-good-sanity', 'nekojs reload server t20w-good.js')
    observe(12, 'active generation liveness markers before any kill')

    print('# [%s] ==== P2 (Leg A): write runaway, full reload -> candidate must be watchdog-killed, old active keeps serving ====' % stamp(), flush=True)
    with open(runaway, 'w', encoding='utf-8', newline='\n') as f:
        f.write("// Ticket 20 AC4 live-watchdog smoke: deliberate infinite loop. With\r\n"
                "// scriptRunawayTimeoutSeconds=3 the wall-clock watchdog interrupts this\r\n"
                "// evaluation; a full reload discards the candidate, a single-file reload\r\n"
                "// (evaluated on the active context) isolates the active runtime.\r\n"
                "console.log('T20W-RUNAWAY entering infinite loop (watchdog should interrupt)')\r\n"
                "while (true) { /* spin forever */ }\r\n")
    print('# [%s] wrote %s' % (stamp(), runaway), flush=True)
    run('reload-full-runaway-1', 'nekojs reload server', quiet=3.0)
    run('error-after-candidate-kill', 'nekojs error')
    observe(12, 'old active generation must still dispatch (candidate was discarded)')

    print('# [%s] ==== P3 (Leg A): second explicit full reload -> fails identically, no auto retry ====' % stamp(), flush=True)
    run('reload-full-runaway-2', 'nekojs reload server', quiet=3.0)
    observe(12, 'old active generation still serving after second failed candidate')

    print('# [%s] ==== P4 (Leg B): single-file reload of the runaway -> watchdog kills the ACTIVE runtime (isolation) ====' % stamp(), flush=True)
    run('file-reload-runaway-active-kill', 'nekojs reload server t20w-runaway.js', quiet=3.0)
    observe(15, 'active is now isolated: dispatch must have STOPPED (no liveness markers)')

    print('# [%s] ==== P5: isolation observables ====' % stamp(), flush=True)
    run('error-while-isolated', 'nekojs error')
    run('file-reload-good-refused-while-isolated', 'nekojs reload server t20w-good.js')

    print('# [%s] ==== P6 (AC4 core): explicit full reload while isolated -> candidate fails again, isolation persists, no auto recovery ====' % stamp(), flush=True)
    run('reload-full-isolated-runaway', 'nekojs reload server', quiet=3.0)
    run('file-reload-good-still-refused', 'nekojs reload server t20w-good.js')
    observe(25, 'no auto recovery window: dead active stays isolated, nothing rebuilds it')

    print('# [%s] ==== P7: remove runaway, explicit full reload must commit and recover ====' % stamp(), flush=True)
    if os.path.exists(runaway):
        os.remove(runaway)
        print('# [%s] removed %s' % (stamp(), runaway), flush=True)
    else:
        print('# [%s] WARN: %s not found' % (stamp(), runaway), flush=True)
    run('reload-full-after-fix', 'nekojs reload server', quiet=3.0)
    run('error-after-recovery', 'nekojs error')
    observe(15, 'recovered generation must dispatch again (liveness markers resume)')

    print('# [%s] all phases done, issuing stop' % stamp(), flush=True)
    rcon.cmd('stop', 10.0, 0.5)
    print('# [%s] stop sent' % stamp(), flush=True)


if __name__ == '__main__':
    main()
