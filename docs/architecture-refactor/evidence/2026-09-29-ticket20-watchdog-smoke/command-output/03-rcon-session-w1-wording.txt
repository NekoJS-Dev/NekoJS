# Ticket 20 W1 live-confirmation driver (temporary tool, embedded verbatim
# into the evidence transcript for reproducibility).
#
# Reduced chain from the watchdog smoke, scoped to W1: the P4 leg
# (single-file reload whose evaluation watchdog-kills the ACTIVE) must now
# answer with failure wording that carries the isolation vocabulary instead
# of "script reload completed", while genuine successes (P1 sanity) and the
# pre-existing-isolation refusal (P5) keep their wording. Recovery (P7)
# restores a healthy active so the server stops cleanly.
import os
import socket
import struct
import sys
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, timeout=90.0, quiet=0.0):
        # Long overall deadline for the FIRST response packet: a reload that a
        # 3s watchdog must interrupt takes seconds before the command answers.
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        deadline = time.time() + timeout
        out = ''
        try:
            while True:
                remaining = deadline - time.time()
                if remaining <= 0:
                    break
                self.s.settimeout(remaining)
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                rid2, _ptype = struct.unpack('<ii', data[:8])
                payload = data[8:-2].decode('utf-8', 'replace')
                if rid2 == self.rid:
                    out += payload
                    if len(payload) < 4000:
                        break
        except socket.timeout:
            pass
        time.sleep(quiet)
        return out.strip()


def wait_for(path, needle, timeout):
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                if needle in f.read():
                    return True
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def stamp():
    return time.strftime('%H:%M:%S')


def main():
    log_path = sys.argv[1]
    scripts_dir = sys.argv[2]
    print('# [%s] waiting for server Done marker in %s' % (stamp(), log_path), flush=True)
    if not wait_for(log_path, 'Done (', 600):
        print('# [%s] FAIL: server never reached Done within 600s' % stamp(), flush=True)
        sys.exit(2)
    print('# [%s] server Done observed, connecting RCON 127.0.0.1:25904' % stamp(), flush=True)
    rcon = Rcon('127.0.0.1', 25904, 't20w1')

    def run(label, cmd, timeout=90.0, quiet=1.5):
        resp = rcon.cmd(cmd, timeout, quiet)
        print('# [%s] RCON [%s] -> %s\n#   resp: %s' % (stamp(), label, cmd, resp if resp else '(empty)'), flush=True)
        return resp

    def observe(seconds, why):
        print('# [%s] observing %ds: %s' % (stamp(), seconds, why), flush=True)
        time.sleep(seconds)
        print('# [%s] observation window ended' % stamp(), flush=True)

    runaway = os.path.join(scripts_dir, 't20w-runaway.js')

    print('# [%s] ==== P1: sanity â€” genuine FILE success keeps its wording ====' % stamp(), flush=True)
    run('error-healthy', 'nekojs error')
    run('file-reload-good-sanity', 'nekojs reload server t20w-good.js')
    observe(10, 'active generation liveness markers before any kill')

    print('# [%s] ==== P4 (W1 leg): single-file reload of the runaway -> watchdog kills the ACTIVE; the answer must be failure wording with the isolation vocabulary ====' % stamp(), flush=True)
    with open(runaway, 'w', encoding='utf-8', newline='\n') as f:
        f.write("// Ticket 20 W1 live confirmation: deliberate infinite loop. With\r\n"
                "// scriptRunawayTimeoutSeconds=3 the wall-clock watchdog interrupts this\r\n"
                "// evaluation on the ACTIVE context (single-file reload), isolating it.\r\n"
                "console.log('T20W-RUNAWAY entering infinite loop (watchdog should interrupt)')\r\n"
                "while (true) { /* spin forever */ }\r\n")
    print('# [%s] wrote %s' % (stamp(), runaway), flush=True)
    run('file-reload-runaway-active-kill', 'nekojs reload server t20w-runaway.js', quiet=3.0)
    observe(15, 'active is now isolated: dispatch must have STOPPED (no liveness markers)')

    print('# [%s] ==== P5: pre-existing isolation refusal keeps its wording ====' % stamp(), flush=True)
    run('error-while-isolated', 'nekojs error')
    run('file-reload-good-refused-while-isolated', 'nekojs reload server t20w-good.js')

    print('# [%s] ==== P7: remove runaway, explicit full reload must commit and recover ====' % stamp(), flush=True)
    if os.path.exists(runaway):
        os.remove(runaway)
        print('# [%s] removed %s' % (stamp(), runaway), flush=True)
    else:
        print('# [%s] WARN: %s not found' % (stamp(), runaway), flush=True)
    run('reload-full-after-fix', 'nekojs reload server', quiet=3.0)
    run('error-after-recovery', 'nekojs error')

    print('# [%s] all phases done, issuing stop' % stamp(), flush=True)
    rcon.cmd('stop', 10.0, 0.5)
    print('# [%s] stop sent' % stamp(), flush=True)


if __name__ == '__main__':
    main()
