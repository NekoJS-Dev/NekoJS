================== RCON driver (temporary tool, embedded for reproducibility) ==================
import socket
import struct
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, wait=1.0):
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        self.s.settimeout(3)
        out = ''
        try:
            while True:
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                out += data[8:-2].decode('utf-8', 'replace')
        except socket.timeout:
            pass
        time.sleep(wait)
        return out.strip()


r = Rcon('127.0.0.1', 25892, 'hotfix36')
print('### bare /nekojs root (F3 usage still intact)')
print(r.cmd('nekojs'))

print('### marker script ran during server start (query reload result surface)')

print('### stopping server')
r.cmd('stop', wait=3.0)   # connection EOF on shutdown is expected
