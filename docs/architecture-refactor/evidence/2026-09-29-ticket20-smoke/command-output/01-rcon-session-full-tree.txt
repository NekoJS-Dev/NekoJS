# Ticket 20 live-command smoke driver: waits for the dedicated server 'Done ('
# marker, then exercises the full /nekojs command tree over RCON as the console
# sender, prints every exchange, removes the deliberate error script, reloads,
# re-checks the error surface, and stops the server cleanly.
import os
import socket
import struct
import sys
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, wait=1.0):
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        self.s.settimeout(3)
        out = ''
        try:
            while True:
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                rid2, _ptype = struct.unpack('<ii', data[:8])
                payload = data[8:-2].decode('utf-8', 'replace')
                if rid2 == self.rid:
                    out += payload
                    if len(payload) < 4000:
                        break
        except socket.timeout:
            pass
        time.sleep(wait)
        return out.strip()


def wait_for(path, needle, timeout):
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                if needle in f.read():
                    return True
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def main():
    log_path = sys.argv[1]
    scripts_dir = sys.argv[2]
    print('# waiting for server Done marker in %s' % log_path, flush=True)
    if not wait_for(log_path, 'Done (', 600):
        print('# FAIL: server never reached Done within 600s', flush=True)
        sys.exit(2)
    print('# server Done observed, connecting RCON', flush=True)
    rcon = Rcon('127.0.0.1', 25892, 'smoke20')

    def run(label, cmd, wait=1.2):
        resp = rcon.cmd(cmd, wait)
        print('# RCON [%s] -> %s\n#   resp: %s' % (label, cmd, resp if resp else '(empty)'), flush=True)
        return resp

    print('# ==== phase B: tree with a deliberate script error present ====', flush=True)
    run('help-tree', 'help nekojs', 1.0)
    run('bare-literal', 'nekojs')
    run('error-with-error', 'nekojs error')
    run('view-all-errors-console', 'nekojs view_all_errors')
    run('reload-default', 'nekojs reload', 8.0)
    run('reload-server', 'nekojs reload server', 8.0)
    run('reload-server-file', 'nekojs reload server t20-single.js', 8.0)
    run('reload-server-file-nested', 'nekojs reload server server_scripts/t20-single.js', 8.0)
    run('reload-startup', 'nekojs reload startup')
    run('reload-startup-file', 'nekojs reload startup t20-single.js')
    run('reload-client', 'nekojs reload client')
    run('reload-client-file', 'nekojs reload client t20-single.js')
    run('reload-test', 'nekojs reload test')
    run('test-command', 'nekojs test')
    run('registry', 'nekojs registry')
    run('registry-stale', 'nekojs registry stale')
    run('packs', 'nekojs packs')
    run('packs-enable-unknown', 'nekojs packs enable no_such_pack')
    run('packs-disable-unknown', 'nekojs packs disable no_such_pack')
    run('hand-console', 'nekojs hand')
    run('inventory-console', 'nekojs inventory')
    run('trust-dedicated', 'nekojs trust 127.0.0.1:25565')
    run('probe-list', 'nekojs probe list')
    run('probe-default', 'nekojs probe', 15.0)
    run('probe-typescript', 'nekojs probe typescript', 15.0)
    run('probe-all', 'nekojs probe all', 20.0)
    run('probe-bad-language', 'nekojs probe cobol', 5.0)
    run('probe-config-reload', 'nekojs probe reload')
    run('probe-enable', 'nekojs probe enable')
    run('probe-disable', 'nekojs probe disable')
    run('probe-while-disabled', 'nekojs probe', 8.0)
    run('probe-reenable', 'nekojs probe enable')
    run('probe-reset-config', 'nekojs probe reset_config', 20.0)

    print('# ==== phase C: remove the deliberate error script, reload, re-check ====', flush=True)
    bad = os.path.join(scripts_dir, 't20-bad.js')
    if os.path.exists(bad):
        os.remove(bad)
        print('# removed %s' % bad, flush=True)
    else:
        print('# WARN: %s not found (already removed?)' % bad, flush=True)
    run('reload-after-fix', 'nekojs reload server', 8.0)
    run('error-after-fix', 'nekojs error')

    print('# all probes sent, issuing stop', flush=True)
    rcon.cmd('stop', 0.5)
    print('# stop sent', flush=True)


if __name__ == '__main__':
    main()
