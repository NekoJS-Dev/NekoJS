// D4: EntityEvents.damagePre return true must negate the damage via setNewDamage(0):
//     the cancelled probe leaves health unchanged and damagePost still fires with 0 damage.
// D5: BlockEvents.randomTick must fire for a vanilla randomly-ticking block (grass_block
//     patch placed via setblock in a forceloaded chunk).
// Every 'T24FIX-' console line is a marker asserted in the verification transcript.

// ---- D4: spider damage probes ----
EntityEvents.damagePre('minecraft:spider', event => {
  console.log('T24FIX-DAMAGE-PRE: original=' + event.originalDamage)
  if (Number(event.originalDamage) === 5) {
    console.log('T24FIX-DAMAGE-CANCEL: cancelling the 5-damage probe (return true)')
    return true
  }
})

EntityEvents.damagePost('minecraft:spider', event => {
  console.log('T24FIX-DAMAGE-POST: original=' + event.originalDamage
    + ' healthDamage=' + event.healthDamage)
})

// ---- D5: vanilla random ticks (grass patch; event fires per randomly-ticking state) ----
let t24fixRandomTickCount = 0
BlockEvents.randomTick(event => {
  if (t24fixRandomTickCount < 3) {
    console.log('T24FIX-RANDOMTICK: pos=' + event.pos + ' block=' + event.block)
  }
  t24fixRandomTickCount++
  if (t24fixRandomTickCount % 25 === 0) {
    console.log('T24FIX-RANDOMTICK-COUNT: ' + t24fixRandomTickCount)
  }
})

================== RCON driver (temporary tool, embedded for reproducibility) ==================
# Ticket 24 D4/D5 fix-verification driver: waits for the dedicated server, forceloads
# the probe area, drives the damage-cancel probes (health via /data get) and the vanilla
# random-tick window (grass patch + random_tick_speed), prints each RCON exchange, then
# stops the server. Temporary tool; embedded in the transcript for reproducibility.
import socket
import struct
import sys
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, wait=1.0):
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        self.s.settimeout(3)
        out = ''
        try:
            while True:
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                rid2, _ptype = struct.unpack('<ii', data[:8])
                payload = data[8:-2].decode('utf-8', 'replace')
                if rid2 == self.rid:
                    out += payload
                    if len(payload) < 4000:
                        break
        except socket.timeout:
            pass
        time.sleep(wait)
        return out.strip()


def wait_for(path, needle, timeout):
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                if needle in f.read():
                    return True
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def main():
    log_path = sys.argv[1]
    print('# waiting for server Done marker in %s' % log_path, flush=True)
    if not wait_for(log_path, 'Done (', 480):
        print('# FAIL: server never reached Done within 480s', flush=True)
        sys.exit(2)
    print('# server Done observed, connecting RCON', flush=True)
    rcon = Rcon('127.0.0.1', 25581, 'nekojs-t24')

    def run(label, cmd, wait=1.0):
        resp = rcon.cmd(cmd, wait)
        print('# RCON [%s] -> %s\n#   resp: %s' % (label, cmd, resp if resp else '(empty)'), flush=True)
        return resp

    run('noon', 'time set noon')
    r = run('forceload', 'forceload add 0 0 40 40', wait=3.0)
    if 'Marked' not in r:
        print('# forceload response unexpected, continuing anyway', flush=True)

    # D5 driver: 26.x renamed the gamerule (randomTickSpeed -> random_tick_speed, verified
    # from GameRules bytecode); grass patch = 25 vanilla randomly-ticking states.
    run('randomtickspeed', 'gamerule random_tick_speed 30')
    for x in range(22, 27):
        for z in range(22, 27):
            rcon.cmd('setblock %d 60 %d minecraft:grass_block' % (x, z), 0.1)
    print('# RCON [grass-patch] -> 25x setblock grass_block 22..26,60,22..26 sent', flush=True)
    time.sleep(2.0)

    # D4 driver: spider subject, cancelled 5-probe then applied 7-probe
    run('spider', 'execute positioned 16 0 16 positioned over world_surface run summon minecraft:spider', wait=2.0)
    run('health-1-before', 'data get entity @e[type=minecraft:spider,limit=1] Health')
    run('damage-5-cancelled', 'execute as @e[type=minecraft:spider,limit=1] run damage @s 5 minecraft:generic')
    run('health-2-after-cancel', 'data get entity @e[type=minecraft:spider,limit=1] Health')
    run('damage-7-applied', 'execute as @e[type=minecraft:spider,limit=1] run damage @s 7 minecraft:generic')
    run('health-3-after-applied', 'data get entity @e[type=minecraft:spider,limit=1] Health')

    # D5 window: random ticks at speed 30 over the grass patch
    print('# randomTick window: 20s', flush=True)
    time.sleep(20)
    run('forceload-remove', 'forceload remove all', 2.0)
    print('# all probes sent, issuing stop', flush=True)
    rcon.cmd('stop', 0.5)
    print('# stop sent', flush=True)


if __name__ == '__main__':
    main()
