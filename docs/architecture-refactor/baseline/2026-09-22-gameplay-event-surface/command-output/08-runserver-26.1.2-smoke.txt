# Ticket 24 AC9 runtime smoke driver, v2: waits for the dedicated server to come up,
# forceloads the probe area (no players keep chunks loaded), drives the event families
# through RCON commands, prints each exchange, then stops the server.
# Evidence transcript: command-output/08-runserver-26.1.2-smoke.txt.
import socket
import struct
import sys
import time


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, wait=1.0):
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        self.s.settimeout(3)
        out = ''
        try:
            while True:
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                rid2, _ptype = struct.unpack('<ii', data[:8])
                payload = data[8:-2].decode('utf-8', 'replace')
                if rid2 == self.rid:
                    out += payload
                    if len(payload) < 4000:
                        break
        except socket.timeout:
            pass
        time.sleep(wait)
        return out.strip()


def wait_for(path, needle, timeout):
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                if needle in f.read():
                    return True
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def main():
    log_path = sys.argv[1]
    print('# waiting for server Done marker in %s' % log_path, flush=True)
    if not wait_for(log_path, 'Done (', 480):
        print('# FAIL: server never reached Done within 480s', flush=True)
        sys.exit(2)
    print('# server Done observed, connecting RCON', flush=True)
    rcon = Rcon('127.0.0.1', 25581, 'nekojs-t24')

    def run(label, cmd, wait=1.0):
        resp = rcon.cmd(cmd, wait)
        print('# RCON [%s] -> %s\n#   resp: %s' % (label, cmd, resp if resp else '(empty)'), flush=True)
        return resp

    run('noon', 'time set noon')
    # CommandEvents: control (must broadcast) vs cancel (must NOT broadcast)
    run('cmd-control', 'say T24-CMD-CONTROL say executed')
    run('cmd-cancelme', 'say t24-cancelme this must not broadcast')
    # keep probe chunks loaded without players
    r = run('forceload', 'forceload add 0 0 40 40', wait=3.0)
    if 'not loaded' in r or 'Marked' not in r:
        print('# forceload response unexpected, continuing anyway', flush=True)

    # BlockEvents.randomTick driver: seed t24smoke spawns in ocean (hopper landed at
    # y=64 = sea level), so place a randomly-ticking grass block explicitly and raise
    # randomTickSpeed so the marker fires within the bounded run.
    run('randomtickspeed', 'gamerule randomTickSpeed 30')
    r = run('grass', 'setblock 24 60 24 minecraft:grass_block')

    # BlockEvents.blockEntityTick driver: hopper ticks every tick once placed
    r = run('hopper', 'execute positioned 8 0 8 positioned over world_surface run setblock ~ ~ ~ minecraft:hopper')
    if 'Changed the block' not in r:
        run('hopper-fallback', 'setblock 8 -60 8 minecraft:hopper')
    # LevelEvents.explosionStart driver: tnt fuse ~4s, kept away from the hopper
    r = run('tnt', 'execute positioned 40 0 40 positioned over world_surface run summon minecraft:tnt')
    if 'Unable' in r or 'not loaded' in r:
        run('tnt-fallback', 'summon minecraft:tnt 40 -58 40')
    # EntityEvents driver: spider as test subject
    r = run('spider', 'execute positioned 16 0 16 positioned over world_surface run summon minecraft:spider')
    if 'Unable' in r or 'not loaded' in r:
        run('spider-fallback', 'summon minecraft:spider 16 -58 16')
    run('health-1', 'data get entity @e[type=minecraft:spider,limit=1] Health')
    run('damage-cancelled-probe', 'execute as @e[type=minecraft:spider,limit=1] run damage @s 5 minecraft:generic')
    run('health-2', 'data get entity @e[type=minecraft:spider,limit=1] Health')
    run('damage-applied-probe', 'execute as @e[type=minecraft:spider,limit=1] run damage @s 7 minecraft:generic')
    run('health-3', 'data get entity @e[type=minecraft:spider,limit=1] Health')
    run('kill-probe', 'execute as @e[type=minecraft:spider,limit=1] run damage @s 1000 minecraft:generic')
    run('spider-gone', 'execute if entity @e[type=minecraft:spider]')
    # give randomTick + tnt some time before shutdown
    time.sleep(10)
    run('forceload-remove', 'forceload remove all', 2.0)
    print('# all probes sent, issuing stop', flush=True)
    rcon.cmd('stop', 0.5)
    print('# stop sent', flush=True)


if __name__ == '__main__':
    main()
