// Every 'T21-' console line is a marker asserted in that transcript.
//
// What this proves beyond the ticket 16 inert declaration face: with
// nekojs/config/engine.toml [dynamicRegistry] enabled = true, the SERVER batch
// declared on DynamicRegistryEvents.dynamicRegistry is committed by the ledger and
// then ACTIVATED by the ticket 21 batch transaction (single node = no remote
// participants, so the coordinator commits without acks) through the real
// NeoForgeDynamicRegistryAdapter surgery path.
//
// INSTANCE distinguishes script generations: the initial server-start load and the
// post-'nekojs reload server' load each re-run this file; their markers must show
// different INSTANCE tokens (reload-driven collect -> joint commit -> per-tick
// pumpActivation -> idempotent re-claim, the entry stays live across the reload).

const INSTANCE = 'gen' + Date.now()
console.log('T21-DEMO-SCRIPT-LOADED ' + INSTANCE)

// Part 1 — declaration surface (the delivered example's batch form, Item + SoundEvent).
// This only STAGES an inert candidate plan; the registry surgery happens later,
// after the ledger commit, inside the activation transaction.
DynamicRegistryEvents.dynamicRegistry(event => {
  event.item('t21:demo_ruby', b => { b.maxStackSize = 16; b.rarity = 'epic' })
  event.soundEvent('t21:demo_boom', b => { b.setFixedRange(16) })
  console.log('T21-DECLARE-STAGED ' + INSTANCE + ': item t21:demo_ruby + soundEvent t21:demo_boom')
})

// Part 2 — LIVE registry observation from the script. The initial batch activates at
// ServerAboutToStart (initial collection pumps directly), the reload batch at the
// next server-tick pump; both precede or trail the first tickPost by at most one
// tick, so the observation retries for a few ticks before declaring failure.
// Item.id() reads the live BuiltInRegistries.ITEM, so a non-null result here is
// direct script-visible proof the adapter's surgery executed.
let ticks = 0
let observed = false
ServerEvents.tickPost(event => {
  ticks++
  if (ticks <= 2) {
    console.log('T21-TICK-' + ticks + ' ' + INSTANCE)
  }
  if (observed) return
  const item = Item.id('t21:demo_ruby')
  if (item === null) {
    if (ticks >= 5) {
      observed = true
      console.log('T21-ACTIVATION-MISSING ' + INSTANCE
        + ': Item.id(t21:demo_ruby) returned null after ' + ticks + ' ticks — the batch did not activate')
    }
    return
  }
  observed = true
  let detail = ''
  try {
    const stack = Item.of('t21:demo_ruby', 2)
    detail = ', registered key ' + String(Item.idOf(item))
      + ', ItemStack ' + String(stack) + ', maxStackSize ' + stack.getMaxStackSize()
  } catch (e) {
    detail = ' (stack detail read failed: ' + e + ')'
  }
  console.log('T21-ACTIVATION-LIVE ' + INSTANCE + ': Item.id resolved t21:demo_ruby at tick ' + ticks + detail)
})

# Ticket 21 AC10 activation demo driver: waits for the dedicated server 'Done ('
# marker, probes the dynamic registry state through RCON, reloads the SERVER script
# domain (exercises the reload-driven collect -> commit -> per-tick pump -> idempotent
# re-activation leg), probes again, then stops the server cleanly.
# Evidence transcript: baseline/2026-09-22-registry-dynamic-sync/command-output/09-runserver-26-1-2-activation.txt
import socket
import struct
import sys
import time

RUN_LOG = sys.argv[1] if len(sys.argv) > 1 else 'versions/26.1.2/run/logs/latest.log'


def pkt(rid, ptype, payload):
    body = struct.pack('<ii', rid, ptype) + payload.encode('utf-8') + b'\x00\x00'
    return struct.pack('<i', len(body)) + body


class Rcon:
    def __init__(self, host, port, password):
        self.s = socket.create_connection((host, port), timeout=10)
        self.rid = 100
        self.s.sendall(pkt(self.rid, 3, password))
        length = struct.unpack('<i', self.recvn(4))[0]
        data = self.recvn(length)
        rid2 = struct.unpack('<i', data[:4])[0]
        if rid2 == -1:
            raise RuntimeError('rcon auth failed')

    def recvn(self, n):
        buf = b''
        while len(buf) < n:
            chunk = self.s.recv(n - len(buf))
            if not chunk:
                raise RuntimeError('rcon eof')
            buf += chunk
        return buf

    def cmd(self, c, wait=1.0):
        self.rid += 1
        self.s.sendall(pkt(self.rid, 2, c))
        self.s.settimeout(3)
        out = ''
        try:
            while True:
                length = struct.unpack('<i', self.recvn(4))[0]
                data = self.recvn(length)
                body = data[8:].decode('utf-8', 'replace')
                if not body:
                    break
                out += body
                if len(out) > 4096:
                    break
        except socket.timeout:
            pass
        time.sleep(wait)
        return out


def wait_for_done(path, timeout_s=420):
    start = time.time()
    while time.time() - start < timeout_s:
        try:
            with open(path, 'r', encoding='utf-8', errors='replace') as f:
                text = f.read()
            if 'Done (' in text:
                return True
            if 'System dump' in text or 'Failed to start' in text:
                return False
        except FileNotFoundError:
            pass
        time.sleep(2)
    return False


def main():
    if not wait_for_done(RUN_LOG):
        print('# ERROR: server Done marker not observed (or server failed)', flush=True)
        sys.exit(2)
    print('# server Done observed, connecting RCON', flush=True)
    rcon = Rcon('127.0.0.1', 25581, 'nekojs-t21')
    probes = [
        ('registry-initial', 'nekojs registry', 1.0),
        ('reload-server', 'nekojs reload server', 5.0),
        ('registry-post-reload', 'nekojs registry', 1.0),
    ]
    for label, cmd, wait in probes:
        resp = rcon.cmd(cmd, wait)
        print('# RCON [%s] -> %s\n#   resp: %s' % (label, cmd, resp if resp else '(empty)'), flush=True)
    print('# all probes sent, issuing stop', flush=True)
    try:
        rcon.cmd('stop', 0.5)
    except RuntimeError as e:
        # 'stop' tears the RCON listener down before a reply is sent (connection eof)
        print('# stop reply absent (expected, listener shuts down first): %s' % e, flush=True)
    print('# stop sent', flush=True)


if __name__ == '__main__':
    main()
